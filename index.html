// Tabla / Sistema de Tierlist de Puntos const PUNTOS_TIERLIST = { "Extreme Demon": 100, "Insane Demon": 75, "Hard Demon": 50, "Medium Demon": 25, "Easy Demon": 10 }; // Función para calcular puntos de un nivel function calcularPuntos(dificultad, progreso) { const puntosBase = PUNTOS_TIERLIST[dificultad] || 0; const porcentaje = parseFloat(progreso) || 100; // Si tiene 100% da todos los puntos, si es un porcentaje menor da la parte proporcional return Math.round((porcentaje / 100) * puntosBase); } document.addEventListener("DOMContentLoaded", () => { cargarNiveles(); }); // Cargar niveles desde el archivo JSON async function cargarNiveles() { try { const response = await fetch('./level-new.json?v=' + Date.now()); if (!response.ok) throw new Error("No se pudo cargar el archivo JSON"); const niveles = await response.json(); renderizarTabla(niveles); } catch (error) { console.error("Error cargando los niveles:", error); } } // Renderizar la tabla con la columna de Puntos (sin botón Eliminar) function renderizarTabla(niveles) { const tbody = document.querySelector("#tabla-niveles tbody"); // Asegúrate de ajustar el ID if (!tbody) return; tbody.innerHTML = ""; niveles.forEach((nivel, index) => { // Si el JSON ya trae un valor manual de puntos usa ese, si no, lo calcula automáticamente const puntos = nivel.puntos !== undefined ? nivel.puntos : calcularPuntos(nivel.dificultad, nivel.progreso); const tr = document.createElement("tr"); tr.innerHTML = ` ${nivel.posicion || index + 1} ${nivel.nombre} ${nivel.creador} ${nivel.dificultad} ${nivel.progreso}% ${Array.isArray(nivel.jugadores) ? nivel.jugadores.join(", ") : nivel.jugadores} +${puntos} pts `; tbody.appendChild(tr); }); }
